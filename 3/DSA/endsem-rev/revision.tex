\label{sec:org33d2ea2}
\begin{itemize}
\item add letters to output
\item kep pushing to stack until stack-top is of equal or higher precedence
\end{itemize}
if eq or higher precedence, keep popping it
\begin{itemize}
\item if ')' pop until '('
\end{itemize}
\subsubsection{postfix to Infix}
\label{sec:orga49ace4}
\begin{itemize}
\item push letters to string stack
\item if operator, push ``(op2 + operator + op1)''
eg: ab+. op1 = b, op2 = a, pushing (a+b)
\end{itemize}
