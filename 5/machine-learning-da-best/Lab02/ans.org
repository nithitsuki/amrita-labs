#+begin_src python :session lab02 :results value :exports both
from sklearn.linear_model import LogisticRegression

X2 = df2[['Candies (#)', 'Mangoes (Kg)', 'Milk Packets (#)']].values
y2 = (df2['Payment (Rs)'] > 200).astype(int).values

clf = LogisticRegression()
clf.fit(X2, y2)

preds = clf.predict(X2)
acc = np.mean(preds == y2)
scores = clf.decision_function(X2)

out = [f"Coef: {clf.coef_}, Intercept: {clf.intercept_[0]:.4f}"]
out.append(f"Predictions: {preds}")
out.append(f"Actual:      {y2}")
out.append(f"Accuracy: {acc:.0%}")
out.append("")
out.append("Decision function:")
for i, s in enumerate(scores):
    out.append(f"  {df2['Customer'].iloc[i]:5s}  {s:+.4f}  ->  {df2['Class'].iloc[i]}")
"\n".join(out)

*** Custom Mean and Variance Functions

#+begin_src python :session lab02 :results value :exports both
def my_mean(x):
    return np.sum(x) / len(x)

def my_var(x):
    mu = my_mean(x)
    return np.sum((x - mu)**2) / len(x)

mu_c = my_mean(price)
var_c = my_var(price)

f"Custom Mean: {mu_c:.4f}, Custom Variance: {var_c:.4f}\nDiff mean: {abs(mu_np-mu_c):.2e}, Diff var: {abs(var_np-var_c):.2e}"
#+end_src

#+RESULTS:
: Custom Mean: 1560.6635, Custom Variance: 58496.4924
: Diff mean: 0.00e+00, Diff var: 0.00e+00

*** Compare Execution Time (10 runs each)

#+begin_src python :session lab02 :results value :exports both
import time

n_runs = 10
t_np = []
for _ in range(n_runs):
    t0 = time.perf_counter()
    np.mean(price); np.var(price)
    t1 = time.perf_counter()
    t_np.append(t1 - t0)

t_custom = []
for _ in range(n_runs):
    t0 = time.perf_counter()
    my_mean(price); my_var(price)
    t1 = time.perf_counter()
    t_custom.append(t1 - t0)

avg_np = np.mean(t_np) * 1e6
avg_cu = np.mean(t_custom) * 1e6

f"NumPy avg: {avg_np:.2f} us, Custom avg: {avg_cu:.2f} us, Ratio: {avg_cu/avg_np:.1f}x"
